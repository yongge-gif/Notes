# filter_by
def get_user_by_username(username):
    user = User.query.filter_by(username).first()  # ❌ 错！

👉 filter_by 必须写字段名

✅ 正确写法
def get_user_by_username(username):
    user = User.query.filter_by(name=username).first()  # ✅
    return user.to_dict() if user else None
